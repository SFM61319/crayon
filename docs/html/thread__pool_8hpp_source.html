23/// @brief Represents a user-created <a
24/// href="https://en.wikipedia.org/wiki/Thread_pool" target="_blank">thread
25/// pool</a>.
26///
27/// You can execute functions explicitly within this @ref ThreadPool using
28/// @ref ThreadPool::submit. @ref ThreadPool::submit executes a closure in one
29/// of the @ref ThreadPool "ThreadPool's" threads.
30///
31/// When this @ref ThreadPool is destructed, that's a signal for the threads it
32/// manages to terminate, they will complete executing any remaining work that
33/// you have spawned, and automatically terminate.
34///
35/// @par Thread safety
36/// @ref ThreadPool::submit, @ref ThreadPool::try_submit, @ref
37/// ThreadPool::shutdown(), and @ref ThreadPool::is_shutting_down may be called
38/// concurrently while this object's lifetime remains active.
39///
40/// Blocking shutdown using @ref ThreadPool::shutdown(bool) with `wait == true`
41/// must not execute concurrently with another blocking shutdown or with
42/// destruction.
43///
44/// Destruction must not race with any operation on this object. Blocking
45/// shutdown and destruction must not be initiated from a worker thread owned by
46/// this same pool.
47///
48/// @par Task execution
49/// Tasks may submit additional tasks to this pool. However, tasks should not
50/// synchronously wait for futures produced by this same pool, as this can
51/// exhaust all workers and deadlock.
52///
53/// Submitted functions and arguments are decay-copied into task storage.
54/// References must be passed explicitly with @ref std::ref and must remain
55/// valid until task execution completes.
56///
57/// Shutdown is cooperative. It does not forcibly interrupt executing user code,
58/// so shutdown may block indefinitely if a task never returns.
59class ThreadPool final {
60public:
61 /// @brief Represents the result of a successful invocation of the @p Function
62 /// with its @p Args.
63 ///
64 /// Requires the @p Function to be invocable with @p Args.
65 ///
66 /// @tparam Function Type of the function called with the @p Args.
67 /// @tparam Args Type of the arguments passed to the @p Function.
68 template <typename Function, typename... Args>
69 requires std::invocable<std::decay_t<Function>, std::decay_t<Args>...>
70 using result_t =
71 std::invoke_result_t<std::decay_t<Function>, std::decay_t<Args>...>;

73 /// @brief Returns an estimate of the default amount of parallelism a program
74 /// should use.
75 ///
76 /// Parallelism is a resource. A given machine provides a certain capacity for
77 /// parallelism, i.e., a bound on the number of computations it can perform
78 /// simultaneously. This number often corresponds to the amount of CPUs a
79 /// computer has, but it may diverge in various cases.
80 ///
81 /// Host environments such as VMs or container orchestrators may want to
82 /// restrict the amount of parallelism made available to programs in them.
83 /// This is often done to limit the potential impact of (unintentionally)
84 /// resource-intensive programs on other programs running on the same machine.
85 ///
86 /// Guaranteed to return a non-zero value.
87 ///
88 /// @return An estimate of the default amount of parallelism a program should
89 /// use.
90 static std::size_t available_parallelism() noexcept;

130 /// @brief Tries to enqueue and submit a function and its arguments for
131 /// execution in a background worker thread in this @ref ThreadPool, returns
132 /// a @ref std::future "future" result on successful submission, and an @ref
133 /// std::nullopt "empty" value otherwise.
134 ///
135 /// Requires the @p function to be invocable with @p args.
136 ///
137 /// An enqueue and submission attempt may fail due to at least one of the
138 /// following reasons:
139 /// - The @ref ThreadPool @ref ThreadPool::is_shutting_down "is shutting
140 /// down".
141 ///
142 /// A failed submission may still consume functions or arguments passed as
143 /// rvalues because task storage is constructed before the pool determines
144 /// whether it can accept the task.
145 ///
146 /// An empty result specifically indicates that the pool rejected the task
147 /// because shutdown had begun. Other failures, including allocation failure
148 /// or exceptions thrown while copying or moving the function or arguments,
149 /// are propagated to the caller.
150 ///
151 /// For a simpler, unwrapped alternative, @see ThreadPool::submit.
152 ///
153 /// @tparam Function Type of the @p function to be called with the @p args.
154 /// @tparam Args Type of the @p args to be passed to the @p function.
155 ///
156 /// @param function The function to be called with the @p args after
157 /// submission.
158 /// @param args The arguments to be passed to the @p function after
159 /// submission.
160 ///
161 /// @return A @ref std::future "future" result on successful submission, and
162 /// an @ref std::nullopt "empty" value otherwise.
163 template <typename Function, typename... Args>
164 requires std::invocable<std::decay_t<Function>, std::decay_t<Args>...>
165 [[nodiscard]]
166 std::optional<std::future<result_t<Function, Args...>>>
167 try_submit(Function &&function, Args &&...args) {
168 // `std::packaged_task` stores the result or exception in shared state that
169 // the returned future can observe.
170 // `std::move_only_function` is used instead of `std::function` because
171 // `std::packaged_task` is move-only.
172 auto task{package_task(std::forward<Function>(function),
173 std::forward<Args>(args)...)};
174
175 auto future{task.get_future()};
176 if (try_enqueue_task_and_notify_one(std::move(task))) {
177 return future;
178 }
179
180 return std::nullopt;
181 }

183 /// @brief Enqueues and submits a function and its arguments for execution in
184 /// a background worker thread in this @ref ThreadPool, returns a @ref
185 /// std::future "future" result on successful submission, and failing
186 /// otherwise.
187 ///
188 /// Requires the @p function to be invocable with @p args.
189 ///
190 /// An enqueue and submission attempt may fail due to at least one of the
191 /// following reasons:
192 /// - The @ref ThreadPool @ref ThreadPool::is_shutting_down "is shutting
193 /// down".
194 ///
195 /// A rejected submission may still consume functions or arguments passed as
196 /// rvalues because task storage is constructed before the pool determines
197 /// whether it can accept the task.
198 ///
199 /// To handle shutdown rejection without an exception, @see
200 /// ThreadPool::try_submit.
201 ///
202 /// @tparam Function Type of the @p function to be called with the @p args.
203 /// @tparam Args Type of the @p args to be passed to the @p function.
204 ///
205 /// @param function The function to be called with the @p args after
206 /// submission.
207 /// @param args The arguments to be passed to the @p function after
208 /// submission.
209 ///
210 /// @return A @ref std::future "future" result on successful submission.
211 /// @throw std::runtime_error If shutdown has begun and the pool no longer
212 /// accepts tasks.
213 /// @throw Any exception propagated while constructing or enqueueing the task,
214 /// including allocation failures and exceptions thrown while copying or
215 /// moving the function or arguments.
216 template <typename Function, typename... Args>
217 requires std::invocable<std::decay_t<Function>, std::decay_t<Args>...>
218 [[nodiscard]]
219 std::future<result_t<Function, Args...>> submit(Function &&function,
220 Args &&...args) {
221 if (auto maybe_future{try_submit(std::forward<Function>(function),
222 std::forward<Args>(args)...)}) {
223 return std::move(*maybe_future);
224 }
225
226 throw std::runtime_error(
227 "crayon::ThreadPool is shutting down and not accepting new tasks");
228 }

230 /// @brief Requests this @ref ThreadPool to gracefully shutdown after draining
231 /// and executing all the queued and inflight tasks, respectively, disallows
232 /// future @ref ThreadPool::submit "task submissions", and returns immediately
233 /// without waiting for this @ref ThreadPool to fully shutdown.
234 ///
235 /// To explicitly control whether to wait until complete shutdown, use @ref
236 /// ThreadPool::shutdown(bool).
237 ///
238 /// This overload is thread-safe and may be called concurrently with task
239 /// submission and with other non-blocking shutdown requests, provided that
240 /// this object's lifetime remains active.
241 ///
242 /// This is an idempotent, irreversible operation. Once shutdown is
243 /// successfully requested, this @ref ThreadPool can no longer be used to
244 /// process tasks. Unless you absolutely need to explicitly request shutdown
245 /// before the end of the scope is reached, prefer letting the compiler shut
246 /// this @ref ThreadPool down on destruction at the end of the scope.
247 void shutdown() noexcept;
248
249 /// @brief Requests this @ref ThreadPool to gracefully shutdown after draining
250 /// and executing all the queued and inflight tasks, respectively, disallows
251 /// future @ref ThreadPool::submit "task submissions", and only waits for this
252 /// @ref ThreadPool to fully shutdown if @p wait is explicitly requested.
253 ///
254 /// When @p wait is `false`, this operation is thread-safe and may be called
255 /// concurrently with task submission and other non-blocking shutdown
256 /// requests while this object's lifetime remains active.
257 ///
258 /// When @p wait is `true`, the caller must ensure that no other blocking
259 /// shutdown or destruction executes concurrently. It must also not be called
260 /// from a worker thread owned by this same pool.
261 ///
262 /// Waiting is cooperative. This function may block indefinitely if an
263 /// executing task never returns or waits on work that cannot make progress.
264 ///
265 /// This is an idempotent, irreversible operation. Once shutdown is
266 /// successfully requested, this @ref ThreadPool can no longer be used to
267 /// process tasks. Unless you absolutely need to explicitly request shutdown
268 /// before the end of the scope is reached, prefer letting the compiler shut
269 /// this @ref ThreadPool down on destruction at the end of the scope.
270 ///
271 /// @param wait Whether to wait until every worker has finished and been
272 /// joined.
273 void shutdown(bool wait) noexcept;
274
275 /// @brief Gracefully shuts down this pool and joins all worker threads.
276 ///
277 /// Queued tasks are drained and executing tasks are allowed to finish before
278 /// destruction completes.
279 ///
280 /// The caller must ensure that no other thread accesses this object once
281 /// destruction begins. Destruction must not be initiated from a worker thread
282 /// owned by this same pool.
283 ~ThreadPool() noexcept;

337 /// @brief Tries to enqueue a nullarized @p task, notifies exactly one worker

342 /// - The @ref ThreadPool @ref ThreadPool::is_shutting_down "is shutting
343 /// down".

345 /// This enqueues the task *and* notifies exactly one worker thread. If you

355 /// @brief Tries to pop a nullarized task, blocking the thread and waiting for
356 /// tasks while sleeping if the @ref ThreadPool::tasks_ "task queue" is empty,
357 /// and returns the task on success and an @ref std::nullopt "empty" value on
358 /// failure.
359 ///
360 /// A pop attempt may fail due to at least one of the following reasons:
361 /// - The @ref ThreadPool::tasks_ "task queue" is empty and the @ref

371 /// value on failure.

409 /// @brief @ref std::packaged_task "Packages" any arbitrary @p function with
410 /// the @p args mapped for nullary invocation and @ref std::promise "promised"
411 /// notification in the @ref std::future "future" and returns it.
412 ///
413 /// The function and arguments are decay-copied into the packaged task and are
414 /// invoked as rvalues. To pass a reference, use @ref std::ref or
415 /// @ref std::cref and ensure the referenced object outlives task execution.
416 ///
417 /// @tparam Function Type of the @p function to be called with the @p args.
418 /// @tparam Args Type of the @p args to be passed to the @p function.
419 ///
420 /// @param function The function to be called with the @p args.
421 /// @param args The arguments to be passed to the @p function.
422 ///
423 /// @return A packaged nullarized task for future invocation and notification.
424 template <typename Function, typename... Args>
425 requires std::invocable<std::decay_t<Function>, std::decay_t<Args>...>
426 [[nodiscard]]
427 static std::packaged_task<result_t<Function, Args...>()>
428 package_task(Function &&function, Args &&...args) {
429 using Result = result_t<Function, Args...>;
430 return std::packaged_task<Result()>{
431 // Workers consume homogeneous `void()` tasks, so bind the callable and
432 // its arguments into a nullary operation.
433 [function{std::forward<Function>(function)},
434 ... args{std::forward<Args>(args)}]() mutable -> Result {
435 return std::invoke(std::move(function), std::move(args)...);
436 }};
437 }
